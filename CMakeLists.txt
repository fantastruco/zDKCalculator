cmake_minimum_required(VERSION 3.20)
project(DokkanDefenseCalculator VERSION 3.12.0 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "The interface requires Windows and Direct3D 11.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(zDKImGuiDirectory ${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui)
add_executable(DokkanDefenseCalculator WIN32 src/zDKWindows.cpp src/zDKUI.h src/zDKUISkin.h src/zDKUIAssets.h src/zDKUINumberFormat.h src/zDKEffects.h src/zDKCalculator.h src/zDKIntegerMath.h src/zDKResources.rc src/zDKVersion.h src/zDKUpdateMetadata.h src/zDKUpdater.h
    ${zDKImGuiDirectory}/imgui.cpp ${zDKImGuiDirectory}/imgui_draw.cpp ${zDKImGuiDirectory}/imgui_tables.cpp ${zDKImGuiDirectory}/imgui_widgets.cpp
    ${zDKImGuiDirectory}/backends/imgui_impl_win32.cpp ${zDKImGuiDirectory}/backends/imgui_impl_dx11.cpp)
target_include_directories(DokkanDefenseCalculator PRIVATE ${zDKImGuiDirectory} ${zDKImGuiDirectory}/backends ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_sources(DokkanDefenseCalculator PRIVATE src/zDKTurnPlan.h src/zDKTurnTimeline.h)
target_compile_definitions(DokkanDefenseCalculator PRIVATE UNICODE _UNICODE)
target_link_libraries(DokkanDefenseCalculator PRIVATE d3d11 dxgi d3dcompiler dwmapi windowscodecs ole32 shell32 comdlg32 user32 gdi32 imm32 winhttp bcrypt)
if(MSVC)
    target_compile_options(DokkanDefenseCalculator PRIVATE /utf-8 /W4)
    target_link_options(DokkanDefenseCalculator PRIVATE /MANIFEST:NO)
endif()
enable_testing()
add_executable(zDKUpdaterTests tests/zDKUpdaterTests.cpp)
target_include_directories(zDKUpdaterTests PRIVATE src)
target_compile_definitions(zDKUpdaterTests PRIVATE UNICODE _UNICODE)
target_link_libraries(zDKUpdaterTests PRIVATE winhttp bcrypt shell32 ole32 user32)
if(MSVC)
    target_compile_options(zDKUpdaterTests PRIVATE /utf-8 /W4)
endif()
add_test(NAME updater COMMAND zDKUpdaterTests)
add_executable(zDKCalculatorTests tests/zDKCalculatorTests.cpp tests/zDKResearchTests.h tests/zDKNumberFormatTests.h tests/zDKRoundingTests.h tests/zDKSupportTests.h)
target_include_directories(zDKCalculatorTests PRIVATE src)
if(MSVC)
    target_compile_options(zDKCalculatorTests PRIVATE /utf-8 /W4)
endif()
add_test(NAME calculator COMMAND zDKCalculatorTests)
target_sources(zDKCalculatorTests PRIVATE tests/zDKTurnTests.h tests/zDKTurnStoreTests.h src/zDKTurnPlan.h src/zDKTurnTimeline.h)
add_test(NAME interface COMMAND DokkanDefenseCalculator --scale 1 --width 540 --height 980 --ui-smoke ${CMAKE_CURRENT_BINARY_DIR}/ui-smoke.log)
set_tests_properties(interface PROPERTIES TIMEOUT 60)
